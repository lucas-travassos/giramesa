=====================================================================
DER ATUALIZADO - GIRAMESA (7 tabelas de dominio + 1 tabela tecnica)
Base: instalacao-inicial.sql + migration 001-historico-integridade.js
Legenda: PK = chave primaria | FK = chave estrangeira | UQ = unico
         NN = nao nulo | N = aceita nulo
=====================================================================

ENTIDADES
---------------------------------------------------------------------
USUARIOS
  usuario_id     INT UNSIGNED   PK, auto incremento
  nome           VARCHAR(100)   NN
  email          VARCHAR(150)   NN, UQ
  senha          VARCHAR(255)   NN (hash bcrypt)
  nivel_acesso   ENUM('garcom','caixa','administrador')   NN
  status         ENUM('ativo','inativo')   NN, padrao 'ativo'
  created_at     DATETIME       NN, padrao agora

MESAS
  mesa_id        INT UNSIGNED   PK, auto incremento
  numero         INT            NN, UQ
  status         ENUM('disponivel','ocupada','caixa','inativa')
                                NN, padrao 'disponivel'
  (inativa = exclusao logica)

CATEGORIAS
  categoria_id   INT UNSIGNED   PK, auto incremento
  nome           VARCHAR(50)    NN, UQ

PRODUTOS
  produto_id     INT UNSIGNED   PK, auto incremento
  categoria_id   INT UNSIGNED   FK -> categorias.categoria_id, NN
  nome           VARCHAR(100)   NN
  descricao      TEXT           NN (texto vazio permitido na aplicacao)
  preco          DECIMAL(8,2)   NN
  status         ENUM('ativo','inativo')   NN, padrao 'ativo'
  created_at     DATETIME       NN, padrao agora

PEDIDOS
  pedido_id        INT UNSIGNED PK, auto incremento
  mesa_id          INT UNSIGNED FK -> mesas.mesa_id, NN
  usuario_id       INT UNSIGNED FK -> usuarios.usuario_id, NN
  data_abertura    DATETIME     NN, padrao agora
  data_fechamento  DATETIME     N (nulo ate a quitacao do pedido)
  status           ENUM('aberto','em_fechamento','finalizado')
                                NN, padrao 'aberto'
  valor_total      DECIMAL(8,2) NN, padrao 0.00

ITENS_PEDIDO
  item_id          INT UNSIGNED PK, auto incremento
  pedido_id        INT UNSIGNED FK -> pedidos.pedido_id, NN
  produto_id       INT UNSIGNED FK -> produtos.produto_id, NN
  quantidade       INT          NN, padrao 1
  preco_unitario   DECIMAL(8,2) NN (preco praticado na venda)
  nome_produto     VARCHAR(100) NN (nome preservado na venda)
  observacao       VARCHAR(255) N

PAGAMENTOS
  pagamento_id     INT UNSIGNED PK, auto incremento
  pedido_id        INT UNSIGNED FK -> pedidos.pedido_id, NN
  forma_pagamento  ENUM('dinheiro','debito','credito','pix')   NN
  valor            DECIMAL(8,2) NN
  data_pagamento   DATETIME     NN, padrao agora


RELACIONAMENTOS
---------------------------------------------------------------------
  CATEGORIAS 1 ---- N PRODUTOS       uma categoria agrupa varios produtos
  MESAS      1 ---- N PEDIDOS        uma mesa tem varios pedidos ao longo do tempo
  USUARIOS   1 ---- N PEDIDOS        um usuario registra varios pedidos
  PEDIDOS    1 ---- N ITENS_PEDIDO   um pedido contem varios itens
  PRODUTOS   1 ---- N ITENS_PEDIDO   um produto aparece em varios itens
  PEDIDOS    1 ---- N PAGAMENTOS     um pedido pode ter varios pagamentos
                                     (divisao de conta parte a parte)

  Observacao: PEDIDOS x PRODUTOS e N:N, resolvido por ITENS_PEDIDO.


INTEGRIDADE E CONTROLE TECNICO
---------------------------------------------------------------------
  ITENS_PEDIDO.nome_produto: snapshot preenchido na venda.
  Itens anteriores: backfill a partir do nome disponivel no cadastro atual.
  Snapshot nao libera renomeacao/categoria de produtos vendidos pela API.
  IX_PEDIDOS_STATUS_FECHAMENTO: pedidos(status, data_fechamento).
  CHECKs: pedidos.valor_total >= 0; pagamentos.valor > 0;
          itens_pedido.quantidade > 0; itens_pedido.preco_unitario > 0;
          produtos.preco > 0.

  SCHEMA_MIGRATIONS (tabela tecnica, sem relacionamento de dominio)
    nome           VARCHAR(100) PK
    checksum       CHAR(64) NN
    aplicada_em    DATETIME NN, padrao agora
